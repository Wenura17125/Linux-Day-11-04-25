Que04:

Sum of prime numbers between 1 and 100 is:1060 



 #!/bin/bash

# Sum of prime numbers between 1 and 100
is_prime() {
  local n=$1
  if (( n < 2 )); then return 1; fi
  for ((i=2; i*i<=n; i++)); do
    if (( n % i == 0 )); then return 1; fi
  done
  return 0
}

sum=0
for ((num=1; num<=100; num++)); do
  if is_prime $num; then
    sum=$((sum + num))
  fi
done
echo "Sum of prime numbers between 1 and 100 is: $sum"
echo

Que03:

Using a while loop to read a file line by line
output:
Number of .txt files: 3

Code:

# Create a sample file list (you can replace this with any real file)
filelist="filelist.txt"
echo -e "notes.txt\nimage.png\ndocument.txt\nreport.txt\nscript.sh" > $filelist

count=0
while read line; do
  if [[ "$line" == *.txt ]]; then
    count=$((count + 1))
  fi
done < "$filelist"

echo "Number of .txt files: $count"
